```tsx
import React from 'react';
import { motion } from 'framer-motion';

const JumpingAliens = () => {
  const colors = [
    'text-red-500',
    'text-blue-500',
    'text-green-500',
    'text-yellow-500',
    'text-purple-500',
    'text-pink-500',
    'text-indigo-500',
    'text-orange-500',
    'text-teal-500',
    'text-cyan-500',
  ];

  return (
    <div className="flex justify-center items-center h-full">
      {[...Array(5)].map((_, index) => {
        const randomColor = colors[Math.floor(Math.random() * colors.length)];
        const randomDuration = Math.random() * 2 + 1;
        
        return (
          <motion.div
            key={index}
            className={`text-4xl mx-2 ${randomColor}`}
            animate={{
              y: [0, -20, 0],
              x: [0, 10, -10, 0],
            }}
            transition={{
              duration: randomDuration,
              repeat: Infinity,
              ease: 'easeInOut',
            }}
          >
            👽
          </motion.div>
        );
      })}
    </div>
  );
};

export default JumpingAliens;
```
