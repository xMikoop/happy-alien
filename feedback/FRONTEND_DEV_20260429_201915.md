```tsx
'use client';

import { Canvas, useFrame } from '@react-three/fiber';
import { Sphere, OrbitControls, PerspectiveCamera } from '@react-three/drei';
import React, { useRef, useMemo } from 'react';
import * as THREE from 'three';

const StarPoints = () => {
  const pointsRef = useRef<THREE.Points>(null);

  // Generowanie pozycji dla 2000 gwiazd w sferze o promieniu 50
  const positions = useMemo(() => {
    const pos = new Float32Array(2000 * 3);
    for (let i = 0; i < 2000; i++) {
      const radius = 50;
      const theta = Math.random() * Math.PI * 2;
      const phi = Math.acos(2 * Math.random() - 1);
      
      const x = radius * Math.sin(phi) * Math.cos(theta);
      const y = radius * Math.sin(phi) * Math.sin(theta);
      const z = radius * Math.cos(phi);

      pos[i * 3] = x;
      pos[i * 3 + 1] = y;
      pos[i * 3 + 2] = z;
    }
    return pos;
  }, []);

  // Kolory gwiazd: biały, niebieski (quasar), rozowy (walszak)
  const colors = useMemo(() => {
    const cols = new Float32Array(2000 * 3);
    for (let i = 0; i < 2000; i++) {
      const rand = Math.random();
      let r = 1, g = 1, b = 1;
      if (rand < 0.33) {
        // Quasar (niebieski)
        r = 0.024; g = 0.714; b = 0.831; // #06b6d4
      } else if (rand < 0.66) {
        // Walszak (rozowy)
        r = 0.980; g = 0.184; b = 0.596; // #fa2f98 (przyblizenie)
      }
      cols[i * 3] = r;
      cols[i * 3 + 1] = g;
      cols[i * 3 + 2] = b;
    }
    return cols;
  }, []);

  const geometry = useMemo(() => {
    const geom = new THREE.BufferGeometry();
    geom.setAttribute('position', new THREE.BufferAttribute(positions, 3));
    geom.setAttribute('color', new THREE.BufferAttribute(colors, 3));
    return geom;
  }, [positions, colors]);

  const material = useMemo(() => {
    return new THREE.PointsMaterial({
      size: 0.5,
      vertexColors: true,
      transparent: true,
      sizeAttenuation: true
    });
  }, []);

  useFrame((state, delta) => {
    if (pointsRef.current) {
      pointsRef.current.rotation.y += delta * 0.05;
    }
  });

  return (
    <points ref={pointsRef} geometry={geometry} material={material} />
  );
};

const Planets = () => {
  const planetRef1 = useRef<THREE.Mesh>(null);
  const planetRef2 = useRef<THREE.Mesh>(null);
  const planetRef3 = useRef<THREE.Mesh>(null);

  useFrame((state, delta) => {
    if (planetRef1.current) {
      planetRef1.current.rotation.y += delta * 0.2;
      planetRef1.current.position.x = 15 * Math.sin(state.clock.elapsedTime * 0.5);
      planetRef1.current.position.z = 15 * Math.cos(state.clock.elapsedTime * 0.5);
    }

    if (planetRef2.current) {
      planetRef2.current.rotation.y += delta * 0.15;
      planetRef2.current.position.x = 25 * Math.sin(state.clock.elapsedTime * 0.3);
      planetRef2.current.position.z = 25 * Math.cos(state.clock.elapsedTime * 0.3);
    }

    if (planetRef3.current) {
      planetRef3.current.rotation.y += delta * 0.1;
      planetRef3.current.position.x = 35 * Math.sin(state.clock.elapsedTime * 0.2);
      planetRef3.current.position.z = 35 * Math.cos(state.clock.elapsedTime * 0.2);
    }
  });

  return (
    <>
      <Sphere args={[1.5, 32, 32]} ref={planetRef1}>
        <meshStandardMaterial color="#6b21a8" /> {/* Nebula */}
      </Sphere>
      <Sphere args={[1, 32, 32]} ref={planetRef2}>
        <meshStandardMaterial color="#fbbf24" /> {/* Star */}
      </Sphere>
      <Sphere args={[1.2, 32, 32]} ref={planetRef3}>
        <meshStandardMaterial color="#06b6d4" /> {/* Quasar */}
      </Sphere>
    </>
  );
};

const GalaxyScene = () => {
  return (
    <>
      <ambientLight intensity={0.3} />
      <pointLight position={[0, 0, 0]} intensity={1} />
      <StarPoints />
      <Planets />
      <PerspectiveCamera makeDefault position={[0, 0, 15]} />
    </>
  );
};

const GalaxyBackground = () => {
  return (
    <Canvas
      className="fixed inset-0 -z-10"
      camera={{ position: [0, 0, 15], fov: 75 }}
      style={{ background: '#0a0a1a' }}
    >
      <GalaxyScene />
    </Canvas>
  );
};

export default GalaxyBackground;
```
